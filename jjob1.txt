#!/bin/bash
#JSUB -q gpu3
#JSUB -J jjob1
#JSUB -n 1
#JSUB -W 0:01

# jjob1.txt - 任务执行脚本1，包含死循环
counter=0
# 使用当前时间作为文件名
TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
DATA_FILE="jdata_${TIMESTAMP}.txt"

# 可配置变量
INITIAL_DELAY=20    # 初始延迟时间（秒）
CHECK_INTERVAL=3    # 检测间隔（秒）
DEBUG=1             # 调试模式开关：1=启用调试，0=禁用调试

# 定义变量
THIS_JOB_NAME="jjob1"
THAT_JOB_NAME="jjob2"
THAT_JOB_FILE="jjob2.txt"
LOG_FILE="jjobs.log"

while true; do
    # 每秒增加计数器
    counter=$((counter + 1))
    
    # 根据DEBUG设置决定是否写入数据文件
    if [ $DEBUG -eq 1 ]; then
        # 将计数写入文件
        echo "$counter" > "$DATA_FILE"
    fi
    
    # 检查jjob2是否运行（初始延迟后，按间隔时间检测）
    if [ $counter -gt $INITIAL_DELAY ] && [ $((counter % $CHECK_INTERVAL)) -eq 0 ]; then
        # 获取jjob2作业ID
        JOB2_ID=$(jjobs -a 2>&1 | grep -E ".*\s+$THAT_JOB_NAME\s+" | awk '{print $1}' | head -1 2>/dev/null)
        
        # 检查jjob2作业是否存在（使用jjobs命令）
        # 作业可能不存在或状态为EXIT/FAILED等停止状态
        if ! jjobs -a 2>&1 | grep -q "$THAT_JOB_NAME"; then
            if [ $DEBUG -eq 1 ]; then
                echo "[$THIS_JOB_NAME] 检测到$THAT_JOB_NAME未运行，立即启动$THAT_JOB_NAME..." >> "$LOG_FILE"
            fi
            # 启动jjob2（使用jsub命令参考jtest.sh的方式）
            jsub -r < "$THAT_JOB_FILE" > /dev/null 2>&1 &
            # 立即检测启动的作业
            sleep 2
            if ! jjobs -a 2>&1 | grep -q "$THAT_JOB_NAME"; then
                if [ $DEBUG -eq 1 ]; then
                    echo "$(date): [$THIS_JOB_NAME] 警告：$THAT_JOB_NAME 启动失败" >> "$LOG_FILE"
                fi
            else
                if [ $DEBUG -eq 1 ]; then
                    echo "$(date): [$THIS_JOB_NAME] $THAT_JOB_NAME 已成功启动" >> "$LOG_FILE"
                fi
            fi
        elif [ -n "$JOB2_ID" ]; then
            # 检查作业是否处于EXIT状态
            STATUS=$(jjobs -w $JOB2_ID 2>/dev/null | awk 'NR>1 {print $3}' 2>/dev/null)
            if [ "$STATUS" = "EXIT" ]; then
                if [ $DEBUG -eq 1 ]; then
                    echo "[$THIS_JOB_NAME] 检测到$THAT_JOB_NAME处于EXIT状态，正在清理并重新启动..." >> "$LOG_FILE"
                fi
                # 清理作业
                jctrl clean $JOB2_ID 2>/dev/null
                if [ $DEBUG -eq 1 ]; then
                    echo "[$THIS_JOB_NAME] $THAT_JOB_NAME 已清理，正在重新启动..." >> "$LOG_FILE"
                fi
                # 重新启动作业
                jsub -r < "$THAT_JOB_FILE" > /dev/null 2>&1 &
                if [ $DEBUG -eq 1 ]; then
                    echo "[$THIS_JOB_NAME] $THAT_JOB_NAME 已重新启动" >> "$LOG_FILE"
                fi
            fi
        fi
    fi
    
    # 等待1秒
    sleep 1
done
